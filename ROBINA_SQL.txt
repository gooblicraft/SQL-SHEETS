-- 1. Create the Vendors table
CREATE TABLE Vendors (
    VendorID VARCHAR(10) PRIMARY KEY,
    VendorName VARCHAR(100),
    ContactNum VARCHAR(20)
);

-- 2. Create the Items table
CREATE TABLE Items (
    ItemID VARCHAR(10) PRIMARY KEY,
    ItemDesc VARCHAR(100),
    VendorID VARCHAR(10)
);

-- 5. DISPLAY THE OUTPUT
-- These commands actually print the tables to your screen

-- Inner Join --
SELECT
    Vendors.VendorName, 
    Items.ItemDesc
FROM Vendors
INNER JOIN Items 
    ON Vendors.VendorID = Items.VendorID;

-- Left Join --    
SELECT 
    Vendors.VendorName, 
    Items.ItemDesc
FROM Vendors
LEFT JOIN Items 
    ON Vendors.VendorID = Items.VendorID;

-- Right Join --
SELECT
  Vendors.VendorName,
  Items.ItemDesc
FROM Vendors

RIGHT JOIN Items
  ON Vendors.VendorID = Items.VendorID;



SELECT 
    Vendors.VendorName, 
    Items.ItemDesc
FROM Vendors
FULL OUTER JOIN Items 
    ON Vendors.VendorID = Items.VendorID;
