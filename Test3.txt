DECLARE @Id INT = 1;
DECLARE @FirstName VARCHAR(50);
DECLARE @LastName VARCHAR(50);
DECLARE @YearsOfService INT;
DECLARE @Status VARCHAR(50);

WHILE @Id <= 10
BEGIN
    -- Fetch data for the current ID
    SELECT @FirstName = FirstName, 
           @LastName = LastName, 
           @YearsOfService = YearsOfService 
    FROM Employees_Tbl 
    WHERE Id = @Id;

    -- Evaluate YearsOfService using CASE logic and UPPER()
    SET @Status = UPPER(
        CASE 
            WHEN @YearsOfService BETWEEN 1 AND 2 THEN 'Trainee'
            WHEN @YearsOfService BETWEEN 3 AND 5 THEN 'Junior'
            WHEN @YearsOfService BETWEEN 6 AND 10 THEN 'Senior'
            WHEN @YearsOfService BETWEEN 11 AND 20 THEN 'Veteran'
            ELSE 'Unassigned'
        END
    );

    -- Update the record
    UPDATE Employees_Tbl
    SET Status = @Status
    WHERE Id = @Id;

    -- Print full name and uppercase status
    PRINT 'Employee: ' + @FirstName + ' ' + @LastName + ' | Status: ' + @Status;

    -- Move to the next ID
    SET @Id = @Id + 1;
END
