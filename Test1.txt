-- Drop table if it already exists to start fresh
IF OBJECT_ID('Employees_Tbl', 'U') IS NOT NULL
    DROP TABLE Employees_Tbl;
GO

-- Create Table directly in the default sandbox database
CREATE TABLE Employees_Tbl (
    Id INT PRIMARY KEY,
    LastName VARCHAR(50),
    FirstName VARCHAR(50),
    YearsOfService INT,
    Department VARCHAR(50),
    Status VARCHAR(50) DEFAULT 'Pending'
);
GO
